Esquema do Banco de Dados (SQL)

Estrutura relacional recomendada em PostgreSQL / Supabase para persistência completa do RPG A Torre das Provações:

-- Criação da tabela de Personagens
CREATE TABLE characters (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES auth.users(id),
    name VARCHAR(100) NOT NULL,
    title VARCHAR(100),
    player VARCHAR(100),
    age INT,
    gender VARCHAR(50),
    appearance TEXT,
    personality TEXT,
    history TEXT,
    objectives TEXT,
    image_url TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Tabela de Atributos e Status
CREATE TABLE character_attributes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    character_id UUID REFERENCES characters(id) ON DELETE CASCADE,
    physical INT DEFAULT 10,
    emotion INT DEFAULT 10,
    mind INT DEFAULT 10,
    hp_current INT DEFAULT 20,
    hp_max INT DEFAULT 20,
    armor_class INT DEFAULT 10,
    dodge_manual BOOLEAN DEFAULT FALSE,
    dodge_val INT DEFAULT 10,
    perception_manual BOOLEAN DEFAULT FALSE,
    perception_val INT DEFAULT 10,
    intuition_manual BOOLEAN DEFAULT FALSE,
    intuition_val INT DEFAULT 10
);

-- Tabela de Campos Personalizados do Personagem
CREATE TABLE custom_fields (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    character_id UUID REFERENCES characters(id) ON DELETE CASCADE,
    field_name VARCHAR(100),
    field_type VARCHAR(50),
    field_value TEXT,
    category VARCHAR(100) DEFAULT 'Geral'
);

-- Tabela de Origens
CREATE TABLE character_origins (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    character_id UUID REFERENCES characters(id) ON DELETE CASCADE,
    origin_name VARCHAR(100),
    description TEXT,
    experience_desc TEXT,
    bonus_value VARCHAR(50) DEFAULT '+2',
    is_custom BOOLEAN DEFAULT FALSE
);

-- Tabela de Habilidades
CREATE TABLE abilities (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    character_id UUID REFERENCES characters(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,
    category VARCHAR(100) DEFAULT 'Geral',
    description TEXT,
    proficiency VARCHAR(50),
    effects TEXT,
    requirements TEXT,
    cost VARCHAR(50),
    cooldown VARCHAR(50),
    notes TEXT,
    icon VARCHAR(50)
);

-- Tabela de Anotações
CREATE TABLE notes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    character_id UUID REFERENCES characters(id) ON DELETE CASCADE,
    category VARCHAR(100) DEFAULT 'Geral',
    title VARCHAR(150),
    content TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
